<script>
    // ---------- DUMMY PAPERS DATA ----------
    const dummyPapers = [
        { id: 1, name: "Physics Paper - March", year: "2023", subject: "Physics", pdfLink: "#" },
        { id: 2, name: "Physics Model Paper", year: "2022", subject: "Physics", pdfLink: "#" },
        { id: 3, name: "Chemistry Past Paper", year: "2023", subject: "Chemistry", pdfLink: "#" },
        { id: 4, name: "Chemistry Practical", year: "2021", subject: "Chemistry", pdfLink: "#" },
        { id: 5, name: "Biology Structured", year: "2022", subject: "Biology", pdfLink: "#" },
        { id: 6, name: "Biology Essay", year: "2023", subject: "Biology", pdfLink: "#" },
        { id: 7, name: "Combined Maths Pure", year: "2023", subject: "Combined Maths", pdfLink: "#" },
        { id: 8, name: "Combined Maths Applied", year: "2022", subject: "Combined Maths", pdfLink: "#" },
        { id: 9, name: "Physics Advanced", year: "2024", subject: "Physics", pdfLink: "#" },
        { id: 10, name: "Chemistry Organic", year: "2024", subject: "Chemistry", pdfLink: "#" },
        { id: 11, name: "Biology Genetics", year: "2021", subject: "Biology", pdfLink: "#" },
        { id: 12, name: "Combined Maths Statistics", year: "2020", subject: "Combined Maths", pdfLink: "#" }
    ];

    // Subjects array
    const subjectsList = ["Physics", "Chemistry", "Biology", "Combined Maths"];

    // Extract unique years for filter
    function getUniqueYears(papers) {
        const years = papers.map(p => p.year);
        return [...new Set(years)].sort((a,b) => b - a);
    }

    // State for filters
    let currentSubjectFilter = "all";
    let currentYearFilter = "all";
    let searchQuery = "";

    // DOM elements
    const papersGrid = document.getElementById("papersGrid");
    const filterSubject = document.getElementById("filterSubject");
    const filterYear = document.getElementById("filterYear");
    const searchInput = document.getElementById("searchPapers");
    const resetBtn = document.getElementById("resetFilters");
    const noPapersMsg = document.getElementById("noPapersMsg");
    const subjectsContainer = document.getElementById("subjectsContainer");

    // render subjects cards + attach quick filter click
    function renderSubjects() {
        subjectsContainer.innerHTML = "";
        subjectsList.forEach(sub => {
            const card = document.createElement("div");
            card.className = "subject-card";
            let icon = "fa-flask";
            if (sub === "Chemistry") icon = "fa-atom";
            else if (sub === "Biology") icon = "fa-dna";
            else if (sub === "Combined Maths") icon = "fa-calculator";
            else icon = "fa-microscope";
            card.innerHTML = `<i class="fas ${icon}"></i><h3>${sub}</h3>`;
            card.addEventListener("click", () => {
                // Quick filter on subject
                filterSubject.value = sub;
                currentSubjectFilter = sub;
                updateFiltersAndRender();
                // smooth scroll to papers
                document.querySelector("#papersGrid").scrollIntoView({ behavior: "smooth", block: "start" });
            });
            subjectsContainer.appendChild(card);
        });
    }

    // Populate year dropdown
    function populateYearDropdown(papersData) {
        const years = getUniqueYears(papersData);
        filterYear.innerHTML = '<option value="all">All Years</option>';
        years.forEach(year => {
            const option = document.createElement("option");
            option.value = year;
            option.textContent = year;
            filterYear.appendChild(option);
        });
    }

    // Filter papers logic (subject, year, search)
    function getFilteredPapers() {
        let filtered = [...dummyPapers];
        if (currentSubjectFilter !== "all") {
            filtered = filtered.filter(p => p.subject === currentSubjectFilter);
        }
        if (currentYearFilter !== "all") {
            filtered = filtered.filter(p => p.year === currentYearFilter);
        }
        if (searchQuery.trim() !== "") {
            const lowerQuery = searchQuery.toLowerCase();
            filtered = filtered.filter(p => 
                p.name.toLowerCase().includes(lowerQuery) || 
                p.subject.toLowerCase().includes(lowerQuery) || 
                p.year.includes(lowerQuery)
            );
        }
        return filtered;
    }

    // Show toast popup with animation for download
    function showDownloadPopup(paperName) {
        const toast = document.getElementById("downloadToast");
        toast.textContent = `📥 Downloading: ${paperName} (sample)`;
        toast.classList.add("show");
        setTimeout(() => {
            toast.classList.remove("show");
        }, 2000);
    }

    // Render papers to grid
    function renderPapers() {
        const filtered = getFilteredPapers();
        if (filtered.length === 0) {
            papersGrid.style.display = "none";
            noPapersMsg.style.display = "block";
            return;
        }
        papersGrid.style.display = "grid";
        noPapersMsg.style.display = "none";
        papersGrid.innerHTML = "";
        filtered.forEach(paper => {
            const card = document.createElement("div");
            card.className = "paper-card";
            card.innerHTML = `
                <div class="paper-info">
                    <div class="paper-year">${paper.year}</div>
                    <div class="paper-subject"><i class="fas fa-graduation-cap"></i> ${paper.subject}</div>
                    <div class="paper-name" style="font-weight:500;">${paper.name}</div>
                </div>
                <button class="download-btn" data-id="${paper.id}" data-name="${paper.name}">
                    <i class="fas fa-download"></i> Download
                </button>
            `;
            const btn = card.querySelector(".download-btn");
            btn.addEventListener("click", (e) => {
                e.stopPropagation();
                // popup animation + dummy placeholder action
                showDownloadPopup(paper.name);
                // simulate download opening placeholder link (optional)
                // we just show popup, but we can also open a new tab if needed (but dummy sample)
                if (paper.pdfLink && paper.pdfLink !== "#") {
                    window.open(paper.pdfLink, '_blank');
                } else {
                    console.log(`Demo: ${paper.name} PDF link placeholder`);
                }
            });
            papersGrid.appendChild(card);
        });
    }

    // update filters from dropdowns and search, then re-render
    function updateFiltersAndRender() {
        currentSubjectFilter = filterSubject.value;
        currentYearFilter = filterYear.value;
        renderPapers();
    }

    // handle search input with debouncing smooth
    let debounceTimer;
    function handleSearchInput() {
        clearTimeout(debounceTimer);
        debounceTimer = setTimeout(() => {
            searchQuery = searchInput.value;
            renderPapers();
        }, 300);
    }

    function resetAllFilters() {
        filterSubject.value = "all";
        filterYear.value = "all";
        searchInput.value = "";
        currentSubjectFilter = "all";
        currentYearFilter = "all";
        searchQuery = "";
        renderPapers();
        // optional: smooth scroll
    }

    // ---------- DARK MODE TOGGLE ----------
    const themeToggle = document.getElementById("themeToggle");
    function initTheme() {
        const savedTheme = localStorage.getItem("sjpepars-theme");
        if (savedTheme === "dark") {
            document.body.classList.add("dark");
            themeToggle.innerHTML = '<i class="fas fa-sun"></i>';
        } else {
            document.body.classList.remove("dark");
            themeToggle.innerHTML = '<i class="fas fa-moon"></i>';
        }
    }
    function toggleTheme() {
        if (document.body.classList.contains("dark")) {
            document.body.classList.remove("dark");
            localStorage.setItem("sjpepars-theme", "light");
            themeToggle.innerHTML = '<i class="fas fa-moon"></i>';
        } else {
            document.body.classList.add("dark");
            localStorage.setItem("sjpepars-theme", "dark");
            themeToggle.innerHTML = '<i class="fas fa-sun"></i>';
        }
    }
    themeToggle.addEventListener("click", toggleTheme);

    // Upload UI only: simple alert with dummy message (no backend)
    const uploadBtnUI = document.getElementById("uploadBtnUI");
    uploadBtnUI.addEventListener("click", () => {
        const paperNameVal = document.getElementById("paperName").value.trim() || "Unnamed Paper";
        const subjectVal = document.getElementById("paperSubject").value;
        const yearVal = document.getElementById("paperYear").value.trim() || "2025";
        if (!paperNameVal) {
            showDownloadPopup("Please enter paper name (UI demo)");
            return;
        }
        // Display a friendly popup that it's UI only
        const toast = document.getElementById("downloadToast");
        toast.textContent = `📄 Demo: "${paperNameVal}" (${subjectVal} ${yearVal}) would be uploaded.`;
        toast.classList.add("show");
        setTimeout(() => toast.classList.remove("show"), 2200);
        // optionally clear fields for better UX
        document.getElementById("paperName").value = "";
        document.getElementById("paperYear").value = "";
    });

    // SMOOTH SCROLLING for any anchor navigation (also subject card trigger already scroll)
    document.querySelectorAll('a[href^="#"]').forEach(anchor => {
        anchor.addEventListener('click', function(e) {
            const href = this.getAttribute('href');
            if(href !== "#" && href !== "") {
                const target = document.querySelector(href);
                if (target) {
                    e.preventDefault();
                    target.scrollIntoView({ behavior: 'smooth' });
                }
            }
        });
    });

    // Event listeners for filters
    filterSubject.addEventListener("change", updateFiltersAndRender);
    filterYear.addEventListener("change", updateFiltersAndRender);
    resetBtn.addEventListener("click", resetAllFilters);
    searchInput.addEventListener("input", handleSearchInput);

    // on window load, populate year dropdown (using dummyPapers initially) + render all
    function init() {
        renderSubjects();
        populateYearDropdown(dummyPapers);
        // set current filters from initial selects
        currentSubjectFilter = filterSubject.value;
        currentYearFilter = filterYear.value;
        renderPapers();
        initTheme();
        // additional smooth scroll behavior for hero search does not require
    }
    init();

    // optional: ensure that year dropdown updates if we ever added papers? but dummy static is fine.
    // also ensure that when reset we sync year dropdown values (already populated)
</script>
</body>
